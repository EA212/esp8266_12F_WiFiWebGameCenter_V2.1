void setup() {
  Serial.begin(115200);
  
  // 初始化SPIFFS
  if (!SPIFFS.begin()) {
    Serial.println("Failed to mount file system");
    return;
  }
  // 配置AP模式
  WiFi.softAP(ssid, password);
  Serial.println();
  Serial.print("AP IP address: ");
  Serial.println(WiFi.softAPIP());
  // 定义路由处理函数
  server.on("/", HTTP_GET, [](AsyncWebServerRequest *request){
    request->send_P(200, "text/html", indexPage); // 使用PROGMEM存储HTML内容
  });
  // 提供获取游戏列表的API
  server.on("/games", HTTP_POST, [](AsyncWebServerRequest *request){
    handleGamesRequest(request);
  });
  // 对于其他所有未找到的请求，尝试读取文件
  server.onNotFound([](AsyncWebServerRequest *request){
    String path = request->url();
    if (handleFileRead(request, path)) return;
    request->send(404, "text/plain", "File Not Found");
  });
  // 启动服务器
  server.begin();
  Serial.println("HTTP server started");
}
void handleGamesRequest(AsyncWebServerRequest *request) {
  DynamicJsonDocument doc(2048); // 增加容量以适应更大的JSON数据
  String requestBody = ""; // 用来存储接收到的body内容
  // 如果有body内容，则将其添加到requestBody字符串中
  if (request->hasParam("plain", true)) { // 检查是否有原始POST数据
    requestBody = request->getParam("plain", true)->value();
  } else {
    Serial.println("No body content received.");
    request->send(400, "text/plain", "Bad Request: No body content");
    return;
  }
  DeserializationError error = deserializeJson(doc, requestBody);
  if (error) {
    Serial.print("deserializeJson() failed: ");
    Serial.println(error.c_str());
    request->send(400, "text/plain", "Bad Request: Invalid JSON");
    return;
  }
  JsonArray cachedGames = doc["games"].as<JsonArray>();
  String gameListJson = "[";
  Dir rootDir = SPIFFS.openDir("/");
  bool firstEntry = true;
  bool printedHeader = false;
  bool hasChanges = false;
  while (rootDir.next()) {
    String fileName = rootDir.fileName();
    if (fileName.startsWith("/")) fileName = fileName.substring(1); // 去掉开头的斜杠
    // 检查是否为文件夹并且包含index.html
    if (SPIFFS.exists(fileName + "/") && SPIFFS.exists(fileName + "/index.html")) {
      bool isNewGame = true;
      for (JsonObject cachedGame : cachedGames) {
        if (cachedGame["name"] == fileName) {
          isNewGame = false;
          break;
        }
      }
      if (isNewGame) {
        if (!firstEntry) gameListJson += ",";
        else firstEntry = false;
        gameListJson += "{\"name\":\"" + fileName + "\",\"url\":\"/" + fileName + "/index.html\"}";

        // 打印文件夹信息到串口，但只打印一次头信息
        if (!printedHeader) {
          Serial.println("Directory structure:");
          printedHeader = true;
        }
        Serial.println("Found new game in folder: " + fileName);
        hasChanges = true;
      }
    }
  }
  gameListJson += "]";
  if (hasChanges) {
    request->send(200, "application/json", gameListJson);
  } else {
    request->send(204, "application/json", ""); // No Content status code
  }
}
bool handleFileRead(AsyncWebServerRequest *request, String path) {
  Serial.println("handleFileRead: " + path);
  // 如果请求的是根路径或文件夹，则默认返回index.html
  if (path.endsWith("/")) {
    path += "index.html";
  }
  // 快速检查文件是否存在，避免不必要的打开操作
  if (!SPIFFS.exists(path)) {
    return false;
  }
  File file = SPIFFS.open(path, "r");
  if (file) {
    // 添加字符集到Content-Type中
    String contentType = getContentType(path);
    if (contentType.startsWith("text/")) {
      contentType += "; charset=UTF-8";
    }
    request->send(SPIFFS, path, contentType);
    file.close();
    return true;
  }

  return false;
}

String getContentType(String filename) {
  if (filename.endsWith(".html")) return "text/html";
  else if (filename.endsWith(".css")) return "text/css";
  else if (filename.endsWith(".js")) return "application/javascript";
  else if (filename.endsWith(".ico")) return "image/x-icon"; // 支持ICO文件
  else if (filename.endsWith(".png")) return "image/png";
  else if (filename.endsWith(".gif")) return "image/gif";
  else if (filename.endsWith(".jpg") || filename.endsWith(".jpeg")) return "image/jpeg";
  else if (filename.endsWith(".svg")) return "image/svg+xml";
  else if (filename.endsWith(".json")) return "application/json";
  else if (filename.endsWith(".mp3")) return "audio/mpeg"; // 支持MP3文件
  // 添加对其他文件类型的处理...
  return "application/octet-stream"; // 默认二进制流类型
}
