#include <ESP8266WiFi.h>
#include <ESP8266WebServer.h>
#include <FS.h>

const char* ssid = "瑜神游戏中心服务器";
const char* password = "";

ESP8266WebServer server(80);

// 存储文件信息
struct FileInfo {
  String path;
  String name;
};
FileInfo fileList[50]; // 最多存储50个文件
int fileCount = 0;
bool isTraversing = false; // 是否正在遍历文件系统

void setup() {
  Serial.begin(115200);

  // 初始化SPIFFS
  if (!SPIFFS.begin()) {
    Serial.println("Failed to mount file system");
    return;
  }

  // 配置AP模式
  WiFi.softAP(ssid, password);
  Serial.println();
  Serial.print("AP IP address: ");
  Serial.println(WiFi.softAPIP());

  // 定义路由处理函数
  server.on("/", handleRoot);
  server.on("/files", handleFiles); // 用于获取文件列表
  server.on("/loading.mp3", []() {
    if (handleFileRead("/loading.mp3")) return;
    server.send(404, "text/plain", "File Not Found");
  });
  server.onNotFound([]() {
    String path = server.uri();
    if (handleFileRead(path)) return;
    server.send(404, "text/plain", "File Not Found");
  });

  // 启动服务器
  server.begin();
  Serial.println("HTTP server started");
}

void loop() {
  server.handleClient();

  // 在没有客户端请求时，逐步遍历文件系统
  if (!isTraversing && fileCount < 50) {
    isTraversing = true;
    traverseFileSystem("/");
    isTraversing = false;
  }
}

// 逐步遍历文件系统
void traverseFileSystem(const char* basePath) {
  Dir dir = SPIFFS.openDir(basePath);
  while (dir.next() && fileCount < 50) {
    File entry = dir.openFile("r");
    String fileName = dir.fileName();
    bool isDir = entry.isDirectory();

    if (isDir) {
      traverseFileSystem(fileName.c_str()); // 递归遍历子文件夹
    } else if (fileName.endsWith("index.html")) {
      fileList[fileCount].path = fileName;
      fileList[fileCount].name = fileName.substring(1, fileName.lastIndexOf('/'));
      fileCount++;
    }
    entry.close();
  }
}

// 读取文件并返回
bool handleFileRead(String path) {
  if (path.endsWith("/")) {
    path += "index.html";
  }
  if (!SPIFFS.exists(path)) {
    return false;
  }
  File file = SPIFFS.open(path, "r");
  if (!file) {
    return false;
  }
  String contentType = getContentType(path);
  if (contentType.startsWith("text/")) {
    contentType += "; charset=UTF-8";
  }
  server.streamFile(file, contentType);
  file.close();
  return true;
}

// 获取文件类型
String getContentType(String filename) {
  if (filename.endsWith(".html")) return "text/html";
  else if (filename.endsWith(".css")) return "text/css";
  else if (filename.endsWith(".js")) return "application/javascript";
  else if (filename.endsWith(".ico")) return "image/x-icon";
  else if (filename.endsWith(".png")) return "image/png";
  else if (filename.endsWith(".gif")) return "image/gif";
  else if (filename.endsWith(".jpg") || filename.endsWith(".jpeg")) return "image/jpeg";
  else if (filename.endsWith(".svg")) return "image/svg+xml";
  else if (filename.endsWith(".json")) return "application/json";
  else if (filename.endsWith(".mp3")) return "audio/mpeg";
  return "application/octet-stream";
}
